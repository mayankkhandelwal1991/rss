Each row sent to your sheet uses these column names as keys. By default we send the filename, all OCR text, and a timestamp.

How to set up the Google Sheet (one-time)
  1. Open a new Google Sheet. Add a header row matching your columns above (e.g. filename | raw_text | timestamp).
  2. Go to Extensions → Apps Script.
  3. Paste the code below, save, then Deploy → New deployment → Web app.
  4. Set Execute as: Me and Who has access: Anyone. Copy the Web App URL into the field above.
function doPost(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const data  = JSON.parse(e.postData.contents);
  const rows  = Array.isArray(data) ? data : [data];

  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn())
                       .getValues()[0];

  rows.forEach(r => {
    const row = headers.map(h => r[h] ?? '');
    sheet.appendRow(row);
  });

  return ContentService
    .createTextOutput(JSON.stringify({ok: true, added: rows.length}))
    .setMimeType(ContentService.MimeType.JSON);
}
